cmake_minimum_required(VERSION 3.20)
set(CMAKE_CXX_COMPILER $ENV{MUSL_BIN}/g++)
project(qwen_cpu CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
add_compile_options(-mavx2 -mfma -mf16c -mavx512f -mavx512bw -mavx512dq -mavx512vl -Wall)
add_library(qwen SHARED qwen.cpp)
target_link_options(qwen PRIVATE -nodefaultlibs)
target_link_libraries(qwen PRIVATE -Wl,-Bstatic -lstdc++ -lgcc -lgcc_eh -lc -Wl,-Bdynamic)
